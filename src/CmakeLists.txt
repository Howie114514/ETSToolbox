set(CMAKE_CXX_STANDARD 23)
file(GLOB SOURCES "*.cpp" "*.c")
#add_compile_definitions(DEBUG_MODE)
add_library(ETSToolbox SHARED ${SOURCES})
set_target_properties(ETSToolbox PROPERTIES 
    CXX_STANDARD 23
    CXX_STANDARD_REQUIRED YES
    MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>DLL"
    OUTPUT_NAME "winmm"
    PREFIX ""
)
if(MSVC AND CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "The winmm proxy stubs use MSVC x86 inline assembly. Configure an x86 build for the 32-bit ETSShell.exe.")
endif()
if(MSVC)
    target_compile_options(ETSToolbox PRIVATE /std:c++latest /EHsc /utf-8)
endif()
find_path(DETOURS_INCLUDE_DIRS "detours/detours.h")
find_library(DETOURS_LIBRARY detours REQUIRED)
target_include_directories(ETSToolbox PRIVATE ${DETOURS_INCLUDE_DIRS})
target_link_libraries(ETSToolbox PRIVATE ${DETOURS_LIBRARY} Shell32)
